        document.getElementById("seconds").innerText = String(seconds).padStart(2, '0');
      } else {
        document.querySelector(".timer-container").innerHTML = "<h3 class='timer-title'>The Grand Pitch Event Has Begun</h3>";
        clearInterval(timerInterval);
      }
    }

    updateCountdown();
    const timerInterval = setInterval(updateCountdown, 1000);

    // Mobile Navigation Controls
    const mobileMenuBtn = document.getElementById('mobileMenuBtn');
    const navLinks = document.getElementById('navLinks');
    if (mobileMenuBtn && navLinks) {
      mobileMenuBtn.addEventListener('click', () => {
        navLinks.classList.toggle('active');
      });
      navLinks.querySelectorAll('a').forEach(link => {
        link.addEventListener('click', () => navLinks.classList.remove('active'));
      });
    }

    // Interactive Web-Lattice Canvas Engine
    const canvas = document.getElementById('webCanvas');
    const ctx = canvas.getContext('2d', {alpha: true});

    let width, height;
    let webNodes = [];
    let pointer = {x: -1000, y: -1000, lastX: null, lastY: null, lastT: null};

    const MAX_LIFE = 900;
    const STEP_PX = 8;
    const MAX_DIST = 140;
    const MAX_NODES = 85;

    // Elegant Corporate Color Array (Crimson, Navy, Tech Cyan, Crisp White)
    const COLORS = [
      {r: 185, g: 28, b: 28},   /* Corporate Crimson */
      {r: 30, g: 58, b: 138},   /* Strategic Navy */
      {r: 0, g: 229, b: 255},   /* Tech Cyan */
      {r: 248, g: 250, b: 252}  /* Pearl White */
    ];

    function resize() {
      const dpr = window.devicePixelRatio || 1;
      width = window.innerWidth;
      height = window.innerHeight;
      canvas.width = width * dpr;
      canvas.height = height * dpr;
      ctx.scale(dpr, dpr);
    }

    resize();
    window.addEventListener('resize', resize, {passive: true});

    function createWebParticles(x, y) {
      const now = performance.now();

      if (pointer.lastX === null) {
        pointer.lastX = x;
        pointer.lastY = y;
        pointer.lastT = now;
      }

      pointer.x = x;
      pointer.y = y;

      const dx = x - pointer.lastX;
      const dy = y - pointer.lastY;
      const dist = Math.hypot(dx, dy);

      const steps = Math.max(1, Math.floor(dist / STEP_PX));

      for (let i = 1; i <= steps; i++) {
        const t = i / steps;
        const px = pointer.lastX + dx * t;
        const py = pointer.lastY + dy * t;

        const color = COLORS[Math.floor(Math.random() * COLORS.length)];

        webNodes.push({
          x: px,
          y: py,
          vx: (Math.random() - 0.5) * 0.7,
          vy: (Math.random() - 0.5) * 0.7,
          t: now,
          baseRadius: 1.8 + Math.random() * 1.6,
          color
        });
      }

      if (webNodes.length > MAX_NODES) {
        webNodes.splice(0, webNodes.length - MAX_NODES);
      }

      pointer.lastX = x;
      pointer.lastY = y;
      pointer.lastT = now;
    }

    window.addEventListener('mousemove', (e) => {
      createWebParticles(e.clientX, e.clientY);
    }, {passive: true});

    window.addEventListener('touchmove', (e) => {
      if (e.touches.length > 0) {
        createWebParticles(e.touches[0].clientX, e.touches[0].clientY);
      }
    }, {passive: true});

    window.addEventListener('touchstart', (e) => {
      if (e.touches.length > 0) {
        pointer.lastX = e.touches[0].clientX;
        pointer.lastY = e.touches[0].clientY;
        createWebParticles(e.touches[0].clientX, e.touches[0].clientY);
      }
    }, {passive: true});

    let lastScrollY = window.scrollY;
    window.addEventListener('scroll', () => {
      const currentScrollY = window.scrollY;
      const deltaY = currentScrollY - lastScrollY;

      if (pointer.x > 0 && pointer.y > 0) {
        createWebParticles(pointer.x, pointer.y + deltaY * 0.15);
      }
      lastScrollY = currentScrollY;
    }, {passive: true});

    let lastFrameTime = performance.now();

    function animate(now) {
      const delta = Math.min((now - lastFrameTime) / 1000, 0.1);
      lastFrameTime = now;

      webNodes = webNodes.filter(p => now - p.t < MAX_LIFE);

      ctx.clearRect(0, 0, width, height);
      ctx.globalCompositeOperation = 'screen';

      const len = webNodes.length;

      // Draw Web-Lattice Connections
      for (let i = 0; i < len; i++) {
        for (let j = i + 1; j < len; j++) {
          const n1 = webNodes[i];
          const n2 = webNodes[j];
          const dx = n1.x - n2.x;
          const dy = n1.y - n2.y;
          const dist = Math.hypot(dx, dy);

          if (dist < MAX_DIST) {
            const age1 = now - n1.t;
            const lifeRatio = 1 - age1 / MAX_LIFE;
            const alpha = (1 - dist / MAX_DIST) * lifeRatio * 0.5;

            ctx.lineWidth = (1 - dist / MAX_DIST) * 1.2;
            const c = n1.color;
            ctx.strokeStyle = `rgba(${c.r}, ${c.g}, ${c.b}, ${alpha})`;

            ctx.beginPath();
            ctx.moveTo(n1.x, n1.y);
            ctx.lineTo(n2.x, n2.y);
            ctx.stroke();
          }
        }
      }

      // Draw Web Nodes
      for (let i = 0; i < len; i++) {
        const p = webNodes[i];
        p.x += p.vx * delta * 60;
        p.y += p.vy * delta * 60;
        p.vx *= 0.96;
        p.vy *= 0.96;

        const age = now - p.t;
        const lifeRatio = 1 - age / MAX_LIFE;
        const alpha = lifeRatio * 0.85;

        const c = p.color;
        ctx.fillStyle = `rgba(${c.r}, ${c.g}, ${c.b}, ${alpha})`;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.baseRadius * lifeRatio, 0, Math.PI * 2);
        ctx.fill();
      }

      // Subtle Pointer Web Glow
      if (pointer.x > 0 && pointer.y > 0) {
        const radialGlow = ctx.createRadialGradient(pointer.x, pointer.y, 0, pointer.x, pointer.y, 80);
        radialGlow.addColorStop(0, 'rgba(0, 229, 255, 0.15)');
        radialGlow.addColorStop(0.5, 'rgba(185, 28, 28, 0.08)');
        radialGlow.addColorStop(1, 'rgba(7, 11, 18, 0)');

        ctx.fillStyle = radialGlow;
        ctx.beginPath();
        ctx.arc(pointer.x, pointer.y, 80, 0, Math.PI * 2);
        ctx.fill();
      }

      requestAnimationFrame(animate);
    }

    requestAnimationFrame(animate);
  </script>
